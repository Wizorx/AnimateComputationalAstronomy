4.1 Funções

Quando precisamos repetir muitas vezes um pedaço de código, faz mais sentido transformalo em uma função: - def -> Definir uma função - return -> O que a função retorna

def soma(a, b):
    return a + b
soma(2,3)
5
# as variaveis dentro da função ficam dentro da função, a menos que você retorne elas

def soma(a, b):
    c = 10
    return a + b + c
soma(2,3)
15
print(c)
---------------------------------------------------------------------------

O que acontece dentro da função fica dentro da função…

Além deles, também temos:

  • args -> Argumentos das funções
  • kwargs -> Argumentos com Keyword
#Vamos supor que queremos fazer a soma, mas o usuário pode escolher quantos valores ele quer colocar.

def soma(a,b):
    return a+b
soma(2,3,5)
---------------------------------------------------------------------------
TypeError                                 Traceback (most recent call last)
Cell In[17], line 1
----> 1 soma(2,3,5)

TypeError: soma() takes 2 positional arguments but 3 were given
# Solução, usar os argumentos

def soma(*args):
    soma = 0
    for valor in args:
        soma += valor
    return soma
soma(2,3)
5
soma(2,3,5,6,7)
23

Em resumo, você pode ter quantos argumentos você quiser, mas sem ter q se preocupar em definir cada um deles.

Podemos inclusive colocar algumas informações extras

def soma(*args, verbose=False):
    soma = 0
    for valor in args:
        if verbose:
            print('somando {:+.1f}'.format(valor))
        soma += valor
    return soma
valor = soma(2,3,5,6,7)

O kwargs é um dicinário! Então como trabalhar com ele?

def soma(*args, **kwargs):
    soma = 0
    if 'verbose' in kwargs:
        verbose = kwargs['verbose']
    for valor in args:
        if verbose:
            print('somando {:+.1f}'.format(valor))
        soma += valor
    return soma

def soma(valor_inicial, *args, **kwargs):
    """Soma os valores adicionados.

    Parâmetros
    ----------
    valor_inicial : `int`, `float`,
        Número inicial que será somado.

    args : `int`, `float`,
        Números que serão somados a variavel valor.

    Kwargs
    ----------
    verbose : `bool`
        Se `True`, irá mostrar informações durante as operações.

    Returns
    -------
    soma : `int`, `float`
        Soma final dos parâmetros de entrada
    """
    soma = valor_inicial
    if 'verbose' in kwargs:
        verbose = kwargs['verbose']
    for valor in args:
        if verbose:
            print('somando {:+.1f}'.format(valor))
        soma += valor
    return soma

class Gatos:
    """ 
    Dockstring blablabla
    """

    def __init__(self, *, numero=0):
        self.contagem = numero
        self.dono = 'Você'

    @property
    def quantos(self):
        return self.contagem

    def adicionar(self, numero):
        """
        adicionar número de gatos
        """
        print('Você tem mais gatos!')
        self.contagem += numero
        return numero

    def mudar_dono(self, name):
        """
        adiciona o nome do dono
        """
        self.dono = name
        return

    def __repr__(self):
        """
        Representação na forma de String
        """
        return '{} tem {} gatos'.format(self.dono, self.contagem)

    def __str__(self):
        """
        Representação na forma de string quando o print(obj) é usado
        """
        return 'você está fazendo o print, e {} tem {} gatos'.format(self.dono, self.contagem)
